%% -------------------------------------------------------
%% Abstract entry command
%% Arguments:
%%   #1  Presentation type label (e.g. Oral, Poster, Demo, Keynote)
%%   #2  Title
%%   #3  Authors  (e.g. Jane Smith^1, John Doe^{1,2})
%%   #4  Affiliations (e.g. ^1 University of X \quad ^2 University of Y)
%%   #5  Abstract body text
%% -------------------------------------------------------
\newcommand{\abstractentry}[5]{%
  \vspace{0.6em}
  {\small\textsc{\color{healtacteal}#1}}\\[2pt]
  {\bfseries #2}\\[2pt]
  {\itshape #3}\\[1pt]
  {\small #4}
  \vspace{4pt}
  \hrule height 0.2pt
  \vspace{4pt}
  {\small #5}
  \vspace{1em}
  \hrule height 0.6pt
  \vspace{0.5em}
}
